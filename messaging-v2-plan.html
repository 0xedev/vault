import Pusher from "pusher";

const pusher = new Pusher({
  appId: process.env.PUSHER_APP_ID!,
  key: process.env.PUSHER_KEY!,
  secret: process.env.PUSHER_SECRET!,
  cluster: process.env.PUSHER_CLUSTER!,
});

export async function POST(req: NextRequest) {
  const auth = await requireUser(req);
  if ("response" in auth) return auth.response;

  const { socket_id, channel_name } = await req.json();
  // channel_name = "private-deal-{escrowId}"

  const escrowId = channel_name.replace("private-deal-", "");
  // Verify user is participant of this escrow
  const rows = await auth.db`
    SELECT id FROM escrows WHERE id = ${escrowId}
    AND (buyer_address = ${auth.user.address} OR seller_address = ${auth.user.address})
    LIMIT 1
  `;
  if (rows.length === 0) return NextResponse.json({}, { status: 403 });

  const authResponse = pusher.authorizeChannel(socket_id, channel_name);
  return NextResponse.json(authResponse);
}

  const auth = await requireUser(req);
  if ("response" in auth) return auth.response;

  // Verify participant
  // ... (same escrow participant check)

  const url = new URL(req.url);
  const limit  = Math.min(parseInt(url.searchParams.get("limit") || "50"), 100);
  const cursor = url.searchParams.get("cursor");

  // Cursor-based: fetch messages older than cursor
  const rows = cursor
    ? await auth.db`
        SELECT * FROM deal_messages
        WHERE escrow_id = ${escrowId}
          AND created_at < ${cursor}::timestamptz
        ORDER BY created_at DESC
        LIMIT ${limit}`
    : await auth.db`
        SELECT * FROM deal_messages
        WHERE escrow_id = ${escrowId}
        ORDER BY created_at DESC
        LIMIT ${limit}`;

  const messages = rows.reverse().map(r => ({
    id: r.id,
    sender: shortAddress(r.sender_address),
    senderAddress: r.sender_address,
    body: decryptMessage(escrowId, r.body),
    messageType: r.message_type || "text",
    imageUrl: r.image_url,
    readAt: r.read_at,
    createdAt: r.created_at,
    me: r.sender_address.toLowerCase() === actorAddress,
  }));

  const nextCursor = rows.length > 0
    ? rows[rows.length - 1].created_at  // oldest in this batch
    : null;

  return NextResponse.json({
    data: messages,
    nextCursor,
    hasMore: rows.length === limit,
  });
}

    { access: "public", token: process.env.BLOB_READ_WRITE_TOKEN }
  );

  return NextResponse.json({ url: blob.url });
}

3.6 HMAC API key auth

// src/lib/api-auth.ts (new file)
import { createHmac, timingSafeEqual } from "crypto";

export function verifyHmacSignature(
  secret: string,
  method: string,
  path: string,
  body: string,
  timestamp: string,
  signature: string
): boolean {
  const payload = `${timestamp}${method}${path}${body}`;
  const expected = createHmac("sha256", secret).update(payload).digest("hex");
  try {
    return timingSafeEqual(
      Buffer.from(expected),
      Buffer.from(signature)
    );
  } catch {
    return false;
  }
}

export async function requireApiKey(req: NextRequest) {
  const apiKey  = req.headers.get("x-vault-api-key");
  const ts      = req.headers.get("x-vault-timestamp");
  const sig     = req.headers.get("x-vault-signature");
  const passphrase = req.headers.get("x-vault-passphrase");

  if (!apiKey || !ts || !sig) return { error: "Missing auth headers" };

  // Replay protection: timestamp must be within ±30s
  if (Math.abs(Date.now() - parseInt(ts)) > 30_000) {
    return { error: "Timestamp expired" };
  }

  // Look up API key
  const db = getDatabase();
  const rows = await db`SELECT * FROM api_keys WHERE api_key = ${apiKey} LIMIT 1`;
  if (rows.length === 0) return { error: "Invalid API key" };

  // HMAC verify
  const body = await req.clone().text(); // need raw body for HMAC
  const valid = verifyHmacSignature(
    rows[0].secret_hash,
    req.method, req.nextUrl.pathname, body, ts, sig!
  );
  if (!valid) return { error: "Invalid signature" };

  // Verify passphrase hash
  const passHash = createHash("sha256").update(passphrase || "").digest("hex");
  if (passHash !== rows[0].passphrase_hash) return { error: "Invalid passphrase" };

  await db`UPDATE api_keys SET last_used_at = NOW() WHERE id = ${rows[0].id}`;

  return { user: { address: rows[0].user_address, role: "user" } };
}

// src/app/(user)/deals/page.tsx — DealRoom component
import Pusher from "pusher-js";
import { useCallback, useEffect, useRef, useState } from "react";

function DealRoom({ deal, onBack, onChanged }) {
  // ... existing state
  const [hasMore, setHasMore] = useState(true);
  const [loadingMore, setLoadingMore] = useState(false);
  const chatRef = useRef<HTMLDivElement>(null);
  const seenRef = useRef<Set<string>>(new Set());

  // ---- Pusher real-time ----
  useEffect(() => {
    const pusher = new Pusher(process.env.NEXT_PUBLIC_PUSHER_KEY!, {
      cluster: process.env.NEXT_PUBLIC_PUSHER_CLUSTER!,
      authEndpoint: "/api/pusher/auth",
      auth: { headers: {} },  // session cookie auto-sent
    });

    const channel = pusher.subscribe(`private-deal-${deal.id}`);

    channel.bind("new-message", (msg: ChatMessage) => {
      setMessages(prev => {
        // Deduplicate — don't add if we already have it (our own sent msg)
        if (prev.some(m => m.id === msg.id)) return prev;
        return [...prev, { ...msg, me: msg.senderAddress === address }];
      });
      // Auto-mark as read if chat is visible
      markAsRead([msg.id]);
    });

    channel.bind("messages-read", (data: { messageIds: string[], readBy: string }) => {
      setMessages(prev => prev.map(m =>
        data.messageIds.includes(m.id) ? { ...m, readAt: new Date().toISOString() } : m
      ));
    });

    channel.bind("stage-change", (data: { stage: string }) => {
      onChanged();  // refresh deal details from server
    });

    return () => {
      pusher.unsubscribe(`private-deal-${deal.id}`);
      pusher.disconnect();
    };
  }, [deal.id, address]);

  // ---- Initial message load + pagination ----
  useEffect(() => {
    loadMessages();
  }, [deal.id, address]);

  const loadMessages = async (cursor?: string) => {
    const params = new URLSearchParams({ limit: "50" });
    if (address) params.set("walletAddress", address);
    if (cursor) params.set("cursor", cursor);

    const res = await fetch(`/api/deals/${deal.id}/messages?${params}`);
    const json = await res.json();

    if (cursor) {
      setMessages(prev => [...json.data.reverse(), ...prev]);
    } else {
      setMessages(json.data || []);
    }
    setHasMore(json.hasMore);
  };

  // ---- Read receipts: mark messages as seen when visible ----
  const markAsRead = async (ids: string[]) => {
    const unread = ids.filter(id => !seenRef.current.has(id));
    if (unread.length === 0) return;
    unread.forEach(id => seenRef.current.add(id));

    await fetch(`/api/deals/${deal.id}/messages/read`, {
      method: "POST",
      credentials: "include",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ messageIds: unread }),
    }).catch(() => {});
  };

  // ---- Image upload ----
  const uploadImage = async (file: File) => {
    const form = new FormData();
    form.set("file", file);
    const res = await fetch(`/api/deals/${deal.id}/messages/image`, {
      method: "POST", credentials: "include", body: form,
    });
    const json = await res.json();
    if (!res.ok) throw new Error(json.error);

    // Send as message
    await fetch(`/api/deals/${deal.id}/messages`, {
      method: "POST", credentials: "include",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        messageType: "image",
        imageUrl: json.url,
        body: "",
      }),
    });
  };
}

// src/lib/notifications.ts
import { sendNotification } from "@farcaster/hub-web";

export async function notifyCounterparty(
  db: DbClient,
  escrow: { id: string; buyer_address: string; seller_address: string },
  senderAddress: string,
  message: string,
) {
  const counterparty = escrow.buyer_address === senderAddress
    ? escrow.seller_address
    : escrow.buyer_address;

  // Check if counterparty has a Farcaster notification token
  const tokens = await db`
    SELECT token FROM notification_tokens
    WHERE user_address = ${counterparty}
      AND platform = 'farcaster'
      AND verified = true
    LIMIT 1
  `;

  if (tokens.length > 0) {
    const fid = parseInt(tokens[0].token);
    await sendNotification({
      fid,
      notification: {
        title: "New message in Deal Room",
        body: message.slice(0, 100),
        targetUrl: `/deals?id=${escrow.id}`,
      },
    }).catch(() => {}); // fire-and-forget
  }

  // Email notification (separate check)
  await sendEmailNotification(db, counterparty, escrow, message);
}

export function decryptMessageAdmin(escrowId: string, ciphertext: string): string {
  // Same decryption — admins use the same MESSAGE_ENCRYPTION_KEY
  // The access control is at the API layer (requireAdmin)

// GET /api/admin/deals/:id/messages
export async function GET(req, { params }) {
  const auth = await requireAdmin(req);     // admin-only
  if ("response" in auth) return auth.response;

  // Verify there's an active dispute for this escrow
  const dispute = await auth.db`
    SELECT id FROM disputes
    WHERE escrow_id = ${escrowId} AND status = 'open'
    LIMIT 1
  `;
  if (dispute.length === 0) {
    return NextResponse.json(
      { error: "No active dispute for this escrow" },
      { status: 403 }
    );
  }

  // Fetch + decrypt all messages
  const rows = await auth.db`
    SELECT * FROM deal_messages WHERE escrow_id = ${escrowId} ORDER BY created_at ASC
  `;

  await writeAudit(
    "ADMIN_VIEWED_MESSAGES", escrowId,
    `Admin viewed all messages for disputed escrow`,
    "admin", auth.user.address
  );

  return NextResponse.json({
    data: rows.map(r => ({
      ...r,
      body: decryptMessage(escrowId, r.body),  // decrypted for admin
    })),
  });
}
